package com.nikita.afk;

import net.fabricmc.api.ClientModInitializer;
import net.fabricmc.fabric.api.client.message.v1.ClientSendMessageEvents;
import net.minecraft.client.MinecraftClient;
import net.minecraft.text.Text;

public class AfkTrackingMod implements ClientModInitializer {

    public static String TELEGRAM_TOKEN = "";
    public static String CHAT_ID = "";

    @Override
    public void onInitializeClient() {
        // Перехватываем отправку сообщений в чат
        ClientSendMessageEvents.ALLOW_CHAT.register(message -> {
            // Если сообщение начинается с обратного слэша
            if (message.startsWith("\\")) {
                String command = message.substring(1).trim(); // Убираем слэш
                handleCommand(command);
                return false; // false = отменить отправку на сервер
            }
            return true; // Обычные сообщения пропускаем на сервер
        });
        System.out.println("✅ AFK Tracking загружен! Используй \\tg_setup, \\tg_status, \\tg_inv");
    }

    private void handleCommand(String command) {
        MinecraftClient client = MinecraftClient.getInstance();
        if (client.player == null) return;

        if (command.startsWith("tg_setup ")) {
            String[] args = command.split(" ");
            if (args.length == 3) {
                TELEGRAM_TOKEN = args[1];
                CHAT_ID = args[2];
                client.player.sendMessage(Text.literal("§a[TG] Токен и ID сохранены!"), false);
            } else {
                client.player.sendMessage(Text.literal("§c[TG] Используй: \\tg_setup <токен> <id>"), false);
            }
        }
        else if (command.equals("tg_status")) {
            client.player.sendMessage(Text.literal("§e[TG] Собираю данные и отправляю..."), false);
            TelegramSender.sendStatus();
        }
        else if (command.equals("tg_inv")) {
            client.player.sendMessage(Text.literal("§e[TG] Отправляю инвентарь..."), false);
            TelegramSender.sendInventory();
        }
    }
}
